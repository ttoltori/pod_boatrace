## 🔧 実装コード例

### Java実装（MlExpectedRecProviderを使用）

```java
public List<MlExpectedRec> getOptimalBets(DBRecord rec) {
    // 戦略A: ベースライン条件
    List<MlExpectedRec> bets = provider.getBetsByCondition(
        rec, 
        "3T",    // 3連単
        1.5,     // maxBexp
        0.05,    // minBprob
        5,       // maxBork
        1        // count (最高スコア1点のみ)
    );
    
    if (bets.isEmpty()) {
        return bets;
    }
    
    // 戦略A+: グレードフィルター
    String grade = rec.getString("grade");
    if (!grade.equals("G1") && !grade.equals("G3")) {
        return new ArrayList<>(); // スキップ
    }
    
    // 戦略A++: 場フィルター（オプション）
    int jyocd = rec.getInt("jyocd");
    int[] topVenues = {6, 10, 13, 16, 17, 20};
    if (!Arrays.stream(topVenues).anyMatch(v -> v == jyocd)) {
        // オプション: 場フィルターを適用しない場合はコメントアウト
        // return new ArrayList<>();
    }
    
    // 期待値範囲の再確認
    MlExpectedRec bet = bets.get(0);
    if (bet.bexp < 1.3 || bet.bexp > 1.5) {
        return new ArrayList<>();
    }
    
    return bets;
}
```
